//Definimos algunas librerias a usar
#define F_CPU 16000000UL
#include <avr/io.h>
#include <util/delay.h>
#include <avr/interrupt.h>

#define BAUD 9600
#define MY_UBRR F_CPU/16/BAUD-1
#define tiempo 24

#define TX_PIN PB0
#define RX_PIN PB1

volatile uint8_t temporizador_activo = 0;
volatile uint8_t segundos = 0;
volatile uint8_t secuencia = 0;   // 0 - Pistón1, 1 - Pistón2, 2 - Vacío, 3 - Reversa (Vacío, Pistón2, Pistón1)
volatile uint8_t tiempo_total = 0; // Contador de tiempo total

#define BUTTON1_PIN PCINT11
#define BUTTON2_PIN PCINT12
#define BUTTON3_PIN PCINT13

// variables globales para saber si los dispositivos están encendidos
volatile uint8_t piston1_encendido = 0;
volatile uint8_t piston2_encendido = 0;
volatile uint8_t vacio_encendido = 0;

// ISR para PCINT0 (botón para activar la secuencia)
ISR(PCINT0_vect) {
	if (!(PINB & (1 << PINB2))) {  // Si el botón está presionado (PB2 en LOW)
		temporizador_activo = 1;    // Activar el temporizador
		TCCR1B = (1 << WGM12) | (1 << CS12) | (1 << CS10);  // Iniciar Timer1
		secuencia = 0;  // Comenzar con el pistón1
		tiempo_total = 0;  // Reiniciar contador de tiempo total
		activarSecuencia();  // Activar el primer paso
	}
		

}

ISR(PCINT1_vect) {
	

	
	if (!(PINC & (1<<BUTTON1_PIN))) {   // Si el botón 1 (PB3) está presionado
		// Activar pistón 1 (PD2)
		PORTD |= PISTON1_PIN;      // Encender pistón 1
		_delay_ms(5000);            // Mantenerlo encendido por un tiempo (ajustable)
		PORTD &= ~PISTON1_PIN;     // Apagar pistón 1
	}

	if (!(PINC & (1<<BUTTON2_PIN))) {   // Si el botón 2 (PB4) está presionado
		// Activar pistón 2 (PD3)
		PORTD |= PISTON2_PIN;      // Encender pistón 2
		_delay_ms(5000);            // Mantenerlo encendido por un tiempo (ajustable)
		PORTD &= ~PISTON2_PIN;     // Apagar pistón 2
	}

	if (!(PINC & (1<<BUTTON3_PIN))) {   // Si el botón 3 (PB5) está presionado
		// Activar vacío (PD4)
		PORTD |= VACIO_PIN;        // Encender vacío
		_delay_ms(5000);            // Mantenerlo encendido por un tiempo (ajustable)
		PORTD &= ~VACIO_PIN;       // Apagar vacío
	}
}
// ISR para Timer1 (CTC)
ISR(TIMER1_COMPA_vect) {
	if (temporizador_activo) {
		segundos++;
		if (segundos >= 2) {  // Después de 2 segundos por paso
			segundos = 0;     // Reiniciar contador de segundos
			tiempo_total += 2; // Sumar 2 segundos al tiempo total
			cambiarSecuencia();  // Cambiar al siguiente paso de la secuencia
			if (tiempo_total >= 30) {  // Después de 30 segundos
				apagarSecuenciaReversa();  // Apagar la secuencia en reversa
			}
		}
	}
}

void configurar_PCINT1(void){
		
		DDRC &= ~((1<<BUTTON1_PIN) | (1<<BUTTON2_PIN) | (1<<BUTTON3_PIN));  // Configurar PC3, PC4, PC5 como entradas
		
		// Activar las resistencias pull-up internas (asumimos botones conectados a GND)
		PORTC |= ((1<<BUTTON1_PIN) | (1<<BUTTON2_PIN) | (1<<BUTTON3_PIN));  // Habilitar pull-ups para PC3, PC4, PC5

		PCICR |= (1 << PCIE1);  
		PCMSK1 |= ((1<<BUTTON1_PIN) | (1<<BUTTON2_PIN) | (1<<BUTTON3_PIN));
}

void configurarPCINT0() {
	PCICR = (1 << PCIE0);               // Habilitar interrupciones PCINT0 (puerto B)
	PCMSK0 = (1 << PCINT2);             // Habilitar interrupción en PCINT2 (PB2)
}

void initSerial() {
	DDRB |= (1 << TX_PIN);
	DDRB &= ~(1 << RX_PIN);
}

void USART_Init() {
	UBRR0H = ((MY_UBRR) >> 8);
	UBRR0L = MY_UBRR;
	UCSR0B = (1 << RXEN0) | (1 << TXEN0);
	UCSR0C = (1 << UCSZ01) | (1 << UCSZ00);

}

void USART_Transmit(unsigned char data) {
	while (!(UCSR0A & (1 << UDRE0)));
	UDR0 = data;
}

unsigned char USART_Receive(void) {
	while (!(UCSR0A & (1 << RXC0)));
	return UDR0;
}

void sendByteGPIO(uint8_t data) { //enviar un byte a través de PB0(tx)
	PORTB &= ~(1 << TX_PIN);
	_delay_us(105);
	
	for(int i = 0; i < 8; i++) {
		if (data & (1 << i)) {
			PORTB |= (1 << TX_PIN);
			} else {
			PORTB &= ~(1 << TX_PIN);
		}
		_delay_us(105);
	}

	PORTB |= (1 << TX_PIN);
	_delay_us(105);
}

uint8_t receiveByteGPIO() {
	uint8_t data = 0;
	
	while (PINB & (1 << RX_PIN));
	
	_delay_us(105);

	for(int i = 0; i < 8; i++) {
		if (PINB & (1 << RX_PIN)) {
			data |= (1 << i);
		}
		_delay_us(105);
	}

	while (!(PINB & (1 << RX_PIN)));

	return data;
}
// Función para configurar Timer1
void configurarTimer1() {
	TCCR1A = 0;                         // Modo CTC
	OCR1A = 15624;                      // Configurar OCR1A para 1 segundo (16 MHz / 1024)
	TIMSK1 = (1 << OCIE1A);             // Habilitar interrupción por comparación
}

int main() {
	initSerial();
	USART_Init();
  // Configuramos interrupciones para los botones de control manual y de secuencia
	configurarPCINT0(); 
	configurar_PCINT1();
	configurarTimer1();  // Configurar Timer1

	// Habilitar interrupciones globales
	sei();


	DDRB |= (1 << TX_PIN);  // TX_PIN como salida
	DDRB &= ~(1 << RX_PIN); // RX_PIN como entrada
	

	while (1) {
		// Control manual basado en los botones
		if (!(PINC & (1 << PC3)) && (PINB & (1 << PC4))) {
			USART_Transmit('5');
			sendByteGPIO('5');
			_delay_ms(50);
			asm("jmp main");
			} else if ((PINB & (1 << PC3)) && !(PINB & (1 << PC4))) {
			USART_Transmit('A');
			sendByteGPIO('A');
			_delay_ms(50);
			asm("jmp main");
			} else if (!(PINB & (1 << PC3)) && !(PINB & (1 << PC4))) {
			USART_Transmit('3');
			sendByteGPIO('3');
			_delay_ms(50);
			asm("jmp main");
			} else if (!(PIND & (1 << PC5))) {
			USART_Transmit('3');
			sendByteGPIO('3');
			_delay_ms(50);
			asm("jmp main");
			} else if (!(PIND & (1 << PB2))) {
			USART_Transmit('9');
			sendByteGPIO('9');
			_delay_ms(50);
			asm("jmp main");
		}
	}
	return 0;
			}
