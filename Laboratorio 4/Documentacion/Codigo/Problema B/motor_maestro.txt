//Definimos algunas librerias a usar en el programa, al igual que algunas variables
#define F_CPU 16000000UL
#include <avr/io.h>
#include <util/delay.h>
#include <avr/interrupt.h>

#define BAUD 9600
#define MY_UBRR F_CPU/16/BAUD-1
#define tiempo 24

#define TX_PIN PB0
#define RX_PIN PB1

// Variables globales
volatile uint8_t led_status = 0; // Rastreo del estado de los LEDs
volatile uint8_t emergency_active = 0; // Estado del apagado de emergencia

ISR(PCINT0_vect) {
	// Verificar si se presionó el botón de apagado de emergencia en PB4
	if (!(PINB & (1 << PB4))) {
		// Apagar todos los LEDs y el motor
		PORTD &= ~((1 << PD2) | (1 << PD3));
		PORTD |= (1<<PD4);
		led_status = 0;

		// Encender el LED en PD4 durante el apagado de emergencia
		PORTD |= (1 << PD4);

		// Marcar apagado de emergencia como activo
		emergency_active = 1;

		// Configurar Timer1 para apagar el LED en PD4 después de 5 segundos
		TCNT1 = 0;                         // Reiniciar contador
		OCR1A = 7812;                      // Comparar cada 5 segundos (16 MHz, prescaler 1024)
		TCCR1B |= (1 << WGM12) | (1 << CS12) | (1 << CS10); // CTC, prescaler 1024
		TIMSK1 |= (1 << OCIE1A);           // Habilitar interrupción por comparación
		return; // Salir de la ISR
	}

	// Si el apagado de emergencia está activo, ignorar otros botones
	if (emergency_active) return;

	// Verificar qué botón fue presionado
	if (!(PINB & (1 << PB2)) && !(led_status & (1 << PB3))) {  // Botón en PB2 y PB3 no está activo
		PORTD |= (1 << PD2);     // Encender LED en PD2
		led_status |= (1 << PB2); // Marcar LED PD2 activo
	}
	if (!(PINB & (1 << PB3)) && !(led_status & (1 << PB2))) {  // Botón en PB3 y PB2 no está activo
		PORTD |= (1 << PD3);     // Encender LED en PD3
		led_status |= (1 << PB3); // Marcar LED PD3 activo
	}

	// Si PB2 o PB3 se activaron, iniciar Timer1
	if (led_status & ((1 << PB2) | (1 << PB3))) {
		// Configurar Timer1 para 5 segundos
		TCNT1 = 0;                         // Reiniciar contador
		OCR1A = 390620;                      // Comparar cada 5 segundos (16 MHz, prescaler 1024)
		TCCR1B |= (1 << WGM12) | (1 << CS12) | (1 << CS10); // CTC, prescaler 1024
		TIMSK1 |= (1 << OCIE1A);           // Habilitar interrupción por comparación
	}
}

ISR(TIMER1_COMPA_vect) {
	// Si el apagado de emergencia está activo, apagar el LED en PD4
	if (emergency_active) {
		PORTD &= ~(1 << PD4);  // Apagar LED en PD4
		emergency_active = 0;  // Resetear el estado de emergencia
		TCCR1B = 0;             // Detener Timer1
		TIMSK1 &= ~(1 << OCIE1A); // Deshabilitar interrupción
		return;
	}

	// Apagar LEDs PB2 y PB3 después de 5 segundos
	if (led_status & (1 << PB2)) {
		PORTD &= ~(1 << PD2);  // Apagar LED en PD2
		led_status &= ~(1 << PB2);
	}
	if (led_status & (1 << PB3)) {
		PORTD &= ~(1 << PD3);  // Apagar LED en PD3
		led_status &= ~(1 << PB3);
	}

	// Si no hay LEDs activos, detener el timer
	if (!(led_status & ((1 << PB2) | (1 << PB3)))) {
		TCCR1B = 0;             // Detener Timer1
		TIMSK1 &= ~(1 << OCIE1A); // Deshabilitar interrupción
	}
}

//Definimos la configuracion de los puertos y de la comunicacion serial con el hc-05 esclavo
void initSerial() {
	DDRB |= (1 << TX_PIN);
	DDRB &= ~(1 << RX_PIN);
}

void USART_Init() {
	UBRR0H = ((MY_UBRR) >> 8);
	UBRR0L = MY_UBRR;
	UCSR0B = (1 << RXEN0) | (1 << TXEN0);
	UCSR0C = (1 << UCSZ01) | (1 << UCSZ00);

}

void USART_Transmit(unsigned char data) {
	while (!(UCSR0A & (1 << UDRE0)));
	UDR0 = data;
}

unsigned char USART_Receive(void) {
	while (!(UCSR0A & (1 << RXC0)));
	return UDR0;
}

void sendByteGPIO(uint8_t data) { //enviar un byte a través de PB0(tx)
	PORTB &= ~(1 << TX_PIN);
	_delay_us(105);
	
	for(int i = 0; i < 8; i++) {
		if (data & (1 << i)) {
			PORTB |= (1 << TX_PIN);
			} else {
			PORTB &= ~(1 << TX_PIN);
		}
		_delay_us(105);
	}

	PORTB |= (1 << TX_PIN);
	_delay_us(105);
}

uint8_t receiveByteGPIO() {
	uint8_t data = 0;
	
	while (PINB & (1 << RX_PIN));
	
	_delay_us(105);

	for(int i = 0; i < 8; i++) {
		if (PINB & (1 << RX_PIN)) {
			data |= (1 << i);
		}
		_delay_us(105);
	}

	while (!(PINB & (1 << RX_PIN)));

	return data;
}


int main() {
//Inicializamos la comunicacion serial

	initSerial();
	USART_Init();





	// Configuramos los botones de parada y de secuencia automatica como entradas
	DDRD &= ~( (1 << PD3) | (1 << PD4)); 
	PORTD |=  (1 << PD3) | (1 << PD4);

// Configurar PB2, PB3 y PB4 como entrada con pull-up
//Botones de cambio de giro
	DDRB &= ~((1 << PB2) | (1 << PB3)); 
	PORTB |= ((1 << PB2) | (1 << PB3));
	// Habilitar interrupciones de cambio de pin para PCINT0 (PB2, PB3, PB4)
	PCICR |= (1 << PCIE0);           // Habilitar grupo PCINT0
	PCMSK0 |= (1 << PCINT2) | (1 << PCINT3) | (1 << PCINT4); // Habilitar PCINT2, PCINT3, PCINT4


	DDRB |= (1 << TX_PIN);  // TX_PIN como salida
	DDRB &= ~(1 << RX_PIN); // RX_PIN como entrada
	
	// Habilitamos interrupciones globales
	sei();

	while (1) {
		//if de control de botones y envio de datos entre microcontroladores
		if (!(PINB & (1 << PB2)) && (PINB & (1 << PB3))) {
			USART_Transmit('9');
			sendByteGPIO('9');
			_delay_ms(50);
			asm("jmp main");
			} else if ((PINB & (1 << PB2)) && !(PINB & (1 << PB3))) {
			USART_Transmit('B');
			sendByteGPIO('B');
			_delay_ms(50);
			asm("jmp main");
			} else if (!(PINB & (1 << PB2)) && !(PINB & (1 << PB3))) {
			USART_Transmit('3');
			sendByteGPIO('3');
			_delay_ms(50);
			asm("jmp main");
			} else if (!(PIND & (1 << PD4))) {
			USART_Transmit('5');
			sendByteGPIO('5');
			_delay_ms(50);
			asm("jmp main");
			} else if (!(PIND & (1 << PD3))) {
			USART_Transmit('3');
			sendByteGPIO('3');
			_delay_ms(50);
			asm("jmp main");
		}
	}
	return 0;
}
