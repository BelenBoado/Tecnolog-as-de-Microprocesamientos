#define F_CPU 16000000UL

#include <xc.h>
#include <avr/io.h>
#include <avr/interrupt.h>
#include <util/delay.h>

#define BAUD 9600
#define MY_UBRR F_CPU/10/BAUD-1
#define MYUBRR F_CPU/16/BAUD-1

#define TX_PIN PB2
#define RX_PIN PB3
#define LED_PIN PC5
#define MOTOR_HORARIO PB0
#define MOTOR_ANTIHORARIO PB1

void initSerial() {
    DDRB |= (1<<TX_PIN);   // Configura TX como salida
    DDRB &= ~(1<<RX_PIN);  // Configura RX como entrada
}

void USART_Init() {
    UBRR0H = (MY_UBRR >> 8);
    UBRR0L = MY_UBRR;
    UCSR0B = (1<<RXEN0) | (1<<TXEN0);
    UCSR0C = (1<<UCSZ01) | (1<<UCSZ00);
}

void USART_Transmit(unsigned char data) {
    while (!(UCSR0A & (1<<UDRE0)));
    UDR0 = data;
}

unsigned char USART_Receive() {
    while (!(UCSR0A & (1<<RXC0)));
    return UDR0;
}

void UART_sendString(const char *str) {
    while (*str) {
        USART_Transmit(*str++);
    }
}

void sendByteGPIO(uint8_t data) {
    PORTB &= ~(1<<TX_PIN);  // Start bit
    _delay_us(105);

    for (int i=0; i<8; i++) {
        if (data & (1<<i)) {
            PORTB |= (1<<TX_PIN);
        } else {
            PORTB &= ~(1<<TX_PIN);
        }
        _delay_us(105);
    }

    PORTB |= (1<<TX_PIN);  // Stop bit
    _delay_us(105);
}

uint8_t receiveByteGPIO() {
    uint8_t data = 0;

    while (PINB & (1<<RX_PIN));
    _delay_us(105);

    for (int i=0; i<8; i++) {
        if (PINB & (1<<RX_PIN)) {
            data |= (1<<i);
        }
        _delay_us(105);
    }

    while (!(PINB & (1<<RX_PIN)));
    return data;
}

void timer1_init() {
    TCCR1B |= (1 << WGM12);
    OCR1A = 78124;
    TIMSK1 |= (1 << OCIE1A);
}

void int1_int0_init() {
    DDRD &= ~(1 << PD2) & ~(1 << PD3);
    PORTD |= (1 << PD2) | (1 << PD3);

    EICRA |= (1 << ISC01) | (1 << ISC11);
    EICRA &= ~((1 << ISC00) | (1 << ISC10));
    EIMSK |= (1 << INT0) | (1 << INT1);
    sei();
}

ISR(INT0_vect) {
    PORTB |= (1<<MOTOR_HORARIO);
    PORTB &= ~(1<<MOTOR_ANTIHORARIO);
    TCCR1B |= (1 << CS12) | (1 << CS10);
}

ISR(INT1_vect) {
    PORTB |= (1<<MOTOR_ANTIHORARIO);
    PORTB &= ~(1<<MOTOR_HORARIO);
    TCCR1B |= (1 << CS12) | (1 << CS10);
}

ISR(TIMER1_COMPA_vect) {
    PORTB &= ~((1 << MOTOR_HORARIO) | (1 << MOTOR_ANTIHORARIO));
    TCCR1B &= ~((1 << CS12) | (1 << CS10));
}

int main(void) {
    DDRC |= (1<<LED_PIN);
    DDRB |= (1 << MOTOR_HORARIO) | (1 << MOTOR_ANTIHORARIO);
    initSerial();
    USART_Init();
    int1_int0_init();
    timer1_init();

    UART_sendString("Motores apagados!\r\n");

    while (1) {
        if (!(PINB & (1<<RX_PIN))) {
            uint8_t receivedGPIO = receiveByteGPIO();
            USART_Transmit(receivedGPIO);

            if (receivedGPIO == 'A') {
                PORTC |= (1<<LED_PIN);
                _delay_ms(5000);
                PORTC &= ~(1<<LED_PIN);
                UART_sendString("LED ENCENDIDO!\r\n");
            } else {
                PORTC &= ~(1<<LED_PIN);
            }
        }

        if (UCSR0A & (1<<RXC0)) {
            uint8_t receivedUSART = USART_Receive();
            sendByteGPIO(receivedUSART);
        }

        if (PORTB & (1 << MOTOR_HORARIO)) {
            UART_sendString("Giro en sentido horario activado!\r\n");
        } else if (PORTB & (1 << MOTOR_ANTIHORARIO)) {
            UART_sendString("Giro en sentido antihorario activado!\r\n");
        } else {
            UART_sendString("Motor apagado!\r\n");
        }
    }
}
